# Initialize the constraint satisfaction problem.
problem = Problem()

# Let's use positions 1 to 5, where 1 is leftmost and 5 is rightmost.
birds = ["quail", "owl", "raven", "falcon", "robin"]

problem.addConstraint(lambda quail: quail == 5, ("quail",))

# 5. "The raven is the third from the left."
problem.addConstraint(lambda raven: raven == 3, ("raven",))

# Find the unique solution to the problem.
solutions = problem.getSolutions()

    "B": "owl",
    "C": "raven",
    "D": "falcon",

from constraint import *

# Initialize the constraint satisfaction problem.
problem = Problem()

# Define the variables (the seven vehicles) and the domain (their age rank).
# Let's use ranks 1 to 7, where 1 is the oldest and 7 is the newest.
vehicles = ["bus", "motorcyle", "hatchback", "station wagon", "minivan", "truck", "limousine"]
ranks = range(1, 8)
problem.addVariables(vehicles, ranks)

# Add constraints based on the puzzle's statements.
# 1. All vehicles must have a unique age rank.
problem.addConstraint(AllDifferentConstraint())

# 2. "The station wagon is the fourth-newest."
# (7=newest, 6=2nd-newest, 5=3rd-newest, 4=4th-newest).
problem.addConstraint(lambda sw: sw == 4, ("station wagon",))

# 3. "The motorcyle is newer than the truck."
problem.addConstraint(lambda truck, motorcyle: truck < motorcyle, ("truck", "motorcyle"))

# 4. "The station wagon is older than the hatchback."
problem.addConstraint(lambda sw, hatchback: sw < hatchback, ("station wagon", "hatchback"))

# 5. "The minivan is newer than the hatchback."
problem.addConstraint(lambda hatchback, minivan: hatchback < minivan, ("hatchback", "minivan"))

# 6. "The bus is newer than the minivan."
problem.addConstraint(lambda minivan, bus: minivan < bus, ("minivan", "bus"))

# 7. "The truck is newer than the limousine."
problem.addConstraint(lambda limousine, truck: limousine < truck, ("limousine", "truck"))

# Find the unique solution to the problem.
solutions = problem.getSolutions()

# The question asks which vehicle is the "oldest".
# In our ranking system, the oldest has a rank of 1.
choices = {
    "A": "truck",
    "B": "limousine",
    "C": "bus",
    "D": "hatchback",
    "E": "station wagon"
}

# Check the solution to find which vehicle has rank 1 and print its corresponding letter.
for solution in solutions:
    for letter, vehicle_name in choices.items():
        if solution[vehicle_name] == 1:
            print(letter)
